Roteiro 2 • Python

Starter Kit Python para Análise & Gráficos

Script executável de 25 linhas (Google Colab ou VS Code) para carregar sua planilha de dados, tabular médias/desvios e gerar gráficos vetoriais em alta resolução (.pdf e .png) para o paper.

analise_experimento.py (Starter Kit) 
import matplotlib.pyplot as plt
import pandas as pd
import numpy as np

# 1. Carregamento de dados (ou CSV da pesquisa)
categorias = ['Portal A', 'Portal B', 'Portal C', 'Portal D', 'Portal E']
pontuacoes = [88.5, 62.0, 94.0, 71.5, 83.0]
desvios    = [3.2, 5.1, 2.0, 4.3, 3.8]

# 2. Estatísticas Descritivas
media_geral = np.mean(pontuacoes)
print(f"Média Geral: {media_geral:.2f} | Desvio: {np.std(pontuacoes):.2f}")

# 3. Gráfico Vetorial para o Artigo SBC
plt.style.use('seaborn-v0_8-whitegrid' if 'seaborn-v0_8-whitegrid' in plt.style.available else 'default')
fig, ax = plt.subplots(figsize=(7, 4.2), dpi=300)

cores = ['#16a34a', '#0284c7', '#7c3aed', '#d97706', '#0891b2']
barras = ax.bar(categorias, pontuacoes, yerr=desvios, capsize=5, color=cores, alpha=0.9, edgecolor='black')
ax.axhline(media_geral, color='#dc2626', linestyle='--', label=f'Média ({media_geral:.1f})')

ax.set_title('Diagnóstico Comparativo de Conformidade Web / Dados', fontsize=11, fontweight='bold')
ax.set_ylabel('Pontuação (0 a 100)', fontsize=10)
ax.set_ylim(0, 110)
ax.legend(loc='upper left')

plt.tight_layout()
plt.savefig('grafico_resultado.png', dpi=300, bbox_inches='tight')
plt.savefig('grafico_resultado.pdf', bbox_inches='tight')
print("-> Gráficos gerados para o Overleaf com sucesso!")
